## Admin CRUD Pattern
The Admin CRUD pattern is the standard approach for creating admin pages in the Fufuni framework. It utilizes the `AdminCrudLayout` component and the `useAdminCrud` hook to provide a unified and efficient way to manage CRUD (Create, Read, Update, Delete) operations.

### Creating a New Admin Page
To create a new admin page, you can use the `AdminCrudLayout` component as the container for your page. This component accepts several props that allow you to customize its behavior and appearance.

```typescript
import { AdminCrudLayout } from '../shared/ui/admin/admin-crud-layout';
import { RowActions } from '../shared/ui/admin/row-actions';
import { useAdminCrud } from '../hooks/use-admin-crud';

// Define the columns for the table
const columns = [
  {
    header: 'ID',
    accessor: 'id',
  },
  {
    header: 'Name',
    accessor: 'name',
  },
  {
    header: 'Actions',
    accessor: 'actions',
  },
];

// Define the data type for the useAdminCrud hook
interface DataType {
  id: string;
  name: string;
}

// Create the admin page component
export function MyAdminPage() {
  const {
    items,
    setItems,
    displayedItems,
    globalFilter,
    setGlobalFilter,
    statusFilter,
    setStatusFilter,
    isModalOpen,
    setIsModalOpen,
    isEditMode,
    editingItem,
    openCreate,
    openEdit,
    closeModal,
  } = useAdminCrud<DataType>({
    filterFn: (item, term) => item.name.toLowerCase().includes(term),
  });

  // Render the AdminCrudLayout component
  return (
    <AdminCrudLayout
      title="My Admin Page"
      addLabel="Add New Item"
      onAdd={openCreate}
      globalFilter={globalFilter}
      onGlobalFilterChange={setGlobalFilter}
      statusFilter={statusFilter}
      onStatusFilterChange={setStatusFilter}
    >
      {/* Render the table with columns and data */}
      <table>
        <thead>
          <tr>
            {columns.map((column) => (
              <th key={column.header}>{column.header}</th>
            ))}
          </tr>
        </thead>
        <tbody>
          {displayedItems.map((item) => (
            <tr key={item.id}>
              <td>{item.id}</td>
              <td>{item.name}</td>
              <td>
                <RowActions
                  onEdit={() => openEdit(item)}
                  onDelete={() => console.log('Delete item', item.id)}
                />
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </AdminCrudLayout>
  );
}
```
