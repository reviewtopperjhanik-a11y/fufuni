```typescript

import { useApi } from '../utils/api';

export const useSecuredApi = () => {
  const api = useApi();
  const jwtToken = api.getJwtToken();
  return {
    get: async (url: string) => {
      const response = await fetch(url, {
        headers: {
          Authorization: `Bearer ${jwtToken}`,
        },
      });
      return response.json();
    },
  };
};
```
