// apps/merchant/src/services/price.ts
export async function getVariantPrice(
  db: Kysely<DB>,
  variantId: string,
  regionId: string
) {
  const row = await db
    .selectFrom('variant_prices')
    .innerJoin('region', 'region.id', 'variant_prices.region_id')
    .where('variant_prices.variant_id', '=', variantId)
    .where('region.id', '=', regionId)
    .select(['variant_prices.amount'])
    .executeTakeFirst();
  return row?.amount ?? 0;
}
