## Step 3: Split Public vs Admin Endpoints
Split the routes into public and admin endpoints. Public endpoints will be registered before auth middleware, while admin endpoints will be registered after.
```typescript
publicApp.openapi(listTagsRoute, async (c) => {
  const db = getDb(c.var.db);
  try {
    const rows = await db.query<TagResponse[]>('SELECT id, name FROM tags');
    return c.json({ items: rows }, 200);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to fetch tags');
  }
});

adminApp.openapi(createTagRoute, async (c) => {
  const db = getDb(c.var.db);
  const body = c.req.valid('json').body;
  try {
    const result = await db.run('INSERT INTO tags (name) VALUES ($1)', [body.name]);
    const tag = await db.query<TagResponse[]>('SELECT id, name FROM tags WHERE id = $1', [result.lastInsertRowId]);
    return c.json(tag[0], 201);
  } catch (error) {
    throw ApiError.badRequest(error instanceof Error ? error.message : 'Failed to create tag');
  }
});
```

## Step 4: Register Routes in index.ts
Register the public route before auth middleware and the admin route after.
```typescript
// apps/merchant/src/index.ts
import { publicApp as publicMyFeature, adminApp as adminMyFeature } from './routes/my-feature';

// ...

app.use('/v1/tags', publicMyFeature);

// ...

app.use(authMiddleware);
app.use('/v1/tags', adminMyFeature);
```

## Step 5: Available RBAC Guards
The available RBAC guards are:
* `adminOnly`: Requires the user to be an admin.
* `superAdminOnly`: Requires the user to be a super admin.
* `databaseAdminOnly`: Requires the user to have database admin permissions.
* `aiAccessOnly`: Requires the user to have AI access permissions.
* `mailAccessOnly`: Requires the user to have mail access permissions.
* `validJwtAuthOnly`: Requires a valid JWT authentication.

// apps/merchant/src/routes/my-feature.ts
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, getDb } from '../types';
import { authMiddleware, adminOnly } from '../middleware/auth';
import { TagResponse, TagListResponse, CreateTagBody } from '../schemas';

const publicApp = new OpenAPIHono();
const adminApp = new OpenAPIHono();

const listTagsRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['Tags'],
  summary: 'List all tags',
  descriptions: 'Returns a list of tags',
  responses: {
    200: {
      content: { 'application/json': { schema: TagListResponse } },
      description: 'List of tags',
    },
    500: {
      content: { 'application/json': { schema: ApiError } },
      description: 'Server error',
    },
  },
});

const createTagRoute = createRoute({
  method: 'post',
  path: '/',
  tags: ['Tags'],
  summary: 'Create a new tag',
  descriptions: 'Creates a new tag',
  responses: {
    201: {
      content: { 'application/json': { schema: TagResponse } },
      description: 'Tag created',
    },
    400: {
      content: { 'application/json': { schema: ApiError } },
      description: 'Invalid request',
    },
    500: {
      content: { 'application/json': { schema: ApiError } },
      description: 'Server error',
    },
  },
  body: CreateTagBody,
});

publicApp.openapi(listTagsRoute, async (c) => {
  const db = getDb(c.var.db);
  try {
    const rows = await db.query<TagResponse[]>('SELECT id, name FROM tags');
    return c.json({ items: rows }, 200);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to fetch tags');
  }

adminApp.openapi(createTagRoute, adminOnly, async (c) => {
  const db = getDb(c.var.db);
  const body = c.req.valid('json').body;
  try {
    const result = await db.run('INSERT INTO tags (name) VALUES ($1)', [body.name]);

    return c.json(tag[0], 201);
  } catch (error) {
    throw ApiError.badRequest(error instanceof Error ? error.message : 'Failed to create tag');
  }
});

export { publicApp as publicMyFeature, adminApp as adminMyFeature };
