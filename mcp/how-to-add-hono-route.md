## How to Add a New API Route
### Step 1: Create a new file for your API route
Create a new file `apps/merchant/src/routes/my-feature.ts` to define your API route.

### Step 2: Add minimal boilerplate
```typescript
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { getDb } from '../db';
import { authMiddleware, adminOnly } from '../middleware/auth';
import { ApiError } from '../types';
import { MyFeatureListResponse, MyFeatureResponse, CreateMyFeatureBody } from '../schemas';

const publicApp = new OpenAPIHono();
const adminApp = new OpenAPIHono();

// Public route: list my features
const listMyFeaturesRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['My Features'],
  summary: 'List all my features',
  responses: {
    200: { 
      content: { 'application/json': { schema: MyFeatureListResponse } },
      description: 'List of my features',
    },
    500: { 
      content: { 'application/json': { schema: ApiError } },
      description: 'Server error',
    },
  },
});

publicApp.openapi(listMyFeaturesRoute, async (c) => {
  const db = getDb(c.var.db);
  try {
    const rows = await db.query<any>('SELECT * FROM my_features');
    return c.json({ items: rows }, 200);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to fetch my features');
  }
});

// Admin route: create my feature
const createMyFeatureRoute = createRoute({
  method: 'post',
  path: '/',
  tags: ['My Features'],
  summary: 'Create a new my feature',
  request: {
    body: CreateMyFeatureBody,
  },
  responses: {
    201: { 
      content: { 'application/json': { schema: MyFeatureResponse } },
      description: 'My feature created',
    },
    400: { 
      content: { 'application/json': { schema: ApiError } },
      description: 'Invalid request',
    },
    500: { 
      content: { 'application/json': { schema: ApiError } },
      description: 'Server error',
    },
  },
});

adminApp.openapi(createMyFeatureRoute, async (c) => {
  const { body } = c.req.valid('json');
  const db = getDb(c.var.db);
  try {
    const result = await db.run('INSERT INTO my_features (name, description) VALUES (?, ?)', [body.name, body.description]);
    return c.json({ id: result.lastInsertRowId, ...body }, 201);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to create my feature');
  }
});

export { publicApp as publicMyFeatures, adminApp as adminMyFeatures };
```

### Step 3: Register the route in index.ts
```typescript
import { publicMyFeatures, adminMyFeatures } from './routes/my-feature';

// Public routes
app.route('/v1/my-features', publicMyFeatures);

// Admin routes
app.route('/v1/my-features', adminMyFeatures);
```

* `mailAccessOnly`: requires mail access
* `validJwtAuthOnly`: requires valid JWT authentication
