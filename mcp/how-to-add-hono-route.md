## How to Add a New API Route

### Step 1: Create a New Route File

Create a new file in the `apps/merchant/src/routes/` directory, e.g., `my-feature.ts`. This file will contain the API route definitions.

### Step 2: Minimal Boilerplate

```typescript
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, type HonoEnv } from '../types';
import { getDb } from '../db';
import { authMiddleware, adminOnly } from '../middleware/auth';

// Create a public and admin router
const publicApp = new OpenAPIHono<HonoEnv>();
const adminApp = new OpenAPIHono<HonoEnv>();
adminApp.use('*', authMiddleware);

// Define a public route (no auth required)
const listMyFeatureRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['My Feature'],
  summary: 'List my feature',
  responses: {
    200: { 
      content: { 'application/json': { schema: z.object({ items: z.array(z.string()) }) } },
      description: 'List of my feature',
    },
  },
});

publicApp.openapi(listMyFeatureRoute, async (c) => {
  // Use getDb() to run queries
  const db = getDb(c.var.db);
  try {
    const rows = await db.query<any>('SELECT * FROM my_feature');
    return c.json({ items: rows }, 200);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to fetch my feature');
  }
});

// Define an admin route (auth required)
const createMyFeatureRoute = createRoute({
  method: 'post',
  path: '/',
  tags: ['My Feature'],
  summary: 'Create my feature (admin)',
  security: [{ bearerAuth: ['admin:store'] }],
  middleware: [adminOnly] as const,
  request: {
    body: {
      content: { 'application/json': { schema: z.object({ name: z.string() }) } },
    },
  },
  responses: {
    201: { 
      content: { 'application/json': { schema: z.object({ id: z.string() }) } },
      description: 'My feature created',
    },
  },
});

adminApp.openapi(createMyFeatureRoute, async (c) => {
  // Use getDb() to run queries
  const db = getDb(c.var.db);
  const body = c.req.valid('json');
  try {
    const result = await db.run('INSERT INTO my_feature (name) VALUES (?)', [body.name]);
    return c.json({ id: result.lastInsertRowid }, 201);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to create my feature');
  }
});

export { publicApp as publicMyFeature, adminApp as adminMyFeature };
```

### Step 3: Register the Route in index.ts

Register the public route before authMiddleware and the admin route after:

```typescript
import { publicMyFeature, adminMyFeature } from './routes/my-feature';

// ...

app.route('/v1/my-feature', publicMyFeature);
app.route('/v1/my-feature', adminMyFeature);
```

### Available RBAC Guards

The following RBAC guards are available:

* `adminOnly`: Requires the `admin:store` permission.
* `superAdminOnly`: Requires the `super:admin` permission.
* `databaseAdminOnly`: Requires the `admin:database` permission.
* `aiAccessOnly`: Requires the `ai:api` permission.
* `mailAccessOnly`: Requires the `mail:api` permission.
* `validJwtAuthOnly`: Requires a valid JWT authentication.

Use these guards to protect your admin routes:

```typescript
import { adminOnly } from '../middleware/auth';

// ...

adminApp.use('*', authMiddleware);
adminApp.openapi(createMyFeatureRoute, async (c) => {
  // ...
}, adminOnly);
```

### Complete Worked Example: Tags Endpoint

Here's a complete example of a "tags" endpoint with GET / (list, public) + POST / (create, adminOnly):

```typescript
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, type HonoEnv } from '../types';
import { getDb } from '../db';
import { authMiddleware, adminOnly } from '../middleware/auth';
import { TagResponse, TagListResponse, CreateTagBody } from '../schemas';

const publicApp = new OpenAPIHono<HonoEnv>();
const adminApp = new OpenAPIHono<HonoEnv>();
adminApp.use('*', authMiddleware);

const listTagsRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['Tags'],
  summary: 'List tags',
  responses: {
    200: { 
      content: { 'application/json': { schema: TagListResponse } },
      description: 'List of tags',
    },
  },
});

publicApp.openapi(listTagsRoute, async (c) => {
  const db = getDb(c.var.db);
  try {
    const rows = await db.query<any>('SELECT * FROM tags');
    return c.json({ items: rows }, 200);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to fetch tags');
  }
});

const createTagRoute = createRoute({
  method: 'post',
  path: '/',
  tags: ['Tags'],
  summary: 'Create tag (admin)',
  security: [{ bearerAuth: ['admin:store'] }],
  middleware: [adminOnly] as const,
  request: {
    body: {
      content: { 'application/json': { schema: CreateTagBody } },
    },
  },
  responses: {
    201: { 
      content: { 'application/json': { schema: TagResponse } },
      description: 'Tag created',
    },
  },
});

adminApp.openapi(createTagRoute, async (c) => {
  const db = getDb(c.var.db);
  const body = c.req.valid('json');
  try {
    const result = await db.run('INSERT INTO tags (name) VALUES (?)', [body.name]);
    const [tag] = await db.query<any>('SELECT * FROM tags WHERE id = ?', [result.lastInsertRowid]);
    return c.json(tag, 201);
  } catch (error) {
    throw ApiError.internalServerError(error instanceof Error ? error.message : 'Failed to create tag');
  }
});

export { publicApp as publicTags, adminApp as adminTags };
```
