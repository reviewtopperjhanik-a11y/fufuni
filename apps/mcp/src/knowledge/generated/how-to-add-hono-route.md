## How to Add a New API Route

Create a new file at `apps/merchant/src/routes/my-feature.ts` with the following structure:

```typescript
// apps/merchant/src/routes/my-feature.ts
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, type HonoEnv } from '../types';
import { getDb } from '../db';
import {
  // Import your Zod schemas from ../schemas
  MyFeatureResponse,
  CreateMyFeatureBody,
  MyFeatureIdParam,
} from '../schemas';

const publicApp = new OpenAPIHono<HonoEnv>();
const adminApp = new OpenAPIHono<HonoEnv>();
```

## Public vs Admin Endpoints

Split endpoints into public and admin sections:

```typescript
// Public routes (no auth)
const publicApp = new OpenAPIHono<HonoEnv>();
publicApp.openapi(/* public routes */);

// Admin routes (with auth)
const adminApp = new OpenAPIHono<HonoEnv>();
adminApp.use('*', authMiddleware);
adminApp.openapi(/* admin-only routes */);

export { publicApp as publicMyFeature, adminApp as adminMyFeature };
```

## Database Operations

Use `getDb()` to run queries:

```typescript
// Read operation
const items = await db.query<MyItem[]>('SELECT * FROM items WHERE status = ?', ['active']);

// Write operation
await db.run(
  'INSERT INTO items (id, name) VALUES (?, ?)',
  [uuid(), 'New Item']
);
```

## Error Handling

Use the `ApiError` class for consistent error responses:

```typescript
try {
  const [item] = await db.query<any>('SELECT * FROM items WHERE id = ?', [id]);
  if (!item) throw ApiError.notFound('Item not found');
  return c.json(item);
} catch (error) {
  if (error instanceof ApiError) throw error;
  throw ApiError.internalServerError('Failed to fetch item');
}
```

## Registering Routes

In `apps/merchant/src/index.ts`, add your route:

```typescript
// Mount public routes BEFORE auth middleware
app.route('/v1/my-feature', publicMyFeature);

// Mount admin routes AFTER auth middleware
app.route('/v1/my-feature', adminMyFeature);
```

## RBAC Guards

Available guards and when to use them:

1. `adminOnly`: For store management (most common admin route)
2. `superAdminOnly`: For system-wide administration
3. `databaseAdminOnly`: For database operations
4. `aiAccessOnly`: For AI-related endpoints
5. `mailAccessOnly`: For email operations
6. `validJwtAuthOnly`: For authenticated user routes

Use them as middleware in route definitions:

```typescript
const adminRoute = createRoute({
  // ...
  middleware: [adminOnly] as const,
  // ...
});
```

## Complete Example: Tags Endpoint

Here's a full implementation of a tags endpoint:

```typescript
// apps/merchant/src/routes/tags.ts
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, uuid, now, type HonoEnv } from '../types';
import { getDb } from '../db';
import { authMiddleware, adminOnly } from '../middleware/auth';
import {
  TagResponse,
  TagListResponse,
  CreateTagBody,
  UpdateTagBody,
  TagIdParam,
} from '../schemas';

const publicApp = new OpenAPIHono<HonoEnv>();
const adminApp = new OpenAPIHono<HonoEnv>();

// Public route: List all tags
const listTagsRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['Tags'],
  summary: 'List all tags',
  responses: {
    200: {
      content: { 'application/json': { schema: TagListResponse } },
      description: 'List of tags',
    },
  },
});

publicApp.openapi(listTagsRoute, async (c) => {
  const db = getDb(c.var.db);
  try {
    const tags = await db.query<{ id: string; name: string; slug: string }>(
      'SELECT id, name, slug FROM tags WHERE status = "active" ORDER BY name ASC'
    );
    return c.json({ items: tags }, 200);
  } catch (error) {
    throw ApiError.internalServerError('Failed to fetch tags');
  }
});

// Admin route: Create a new tag
const createTagRoute = createRoute({
  method: 'post',
  path: '/',
  tags: ['Tags'],
  summary: 'Create a new tag (admin)',
  security: [{ bearerAuth: ['admin:store'] }],
  middleware: [adminOnly] as const,
  request: {
    body: {
      content: { 'application/json': { schema: CreateTagBody } },
    },
  },
  responses: {
    201: {
      content: { 'application/json': { schema: TagResponse } },
      description: 'Tag created',
    },
    400: {
      content: { 'application/json': { schema: ErrorResponse } },
      description: 'Invalid input',
    },
  },
});

adminApp.openapi(createTagRoute, async (c) => {
  const body = c.req.valid('json');
  const db = getDb(c.var.db);

  try {
    // Check for duplicate slug
    const [existing] = await db.query<{ id: string }>(
      'SELECT id FROM tags WHERE slug = ?',
      [body.slug]
    );
    if (existing) {
      throw ApiError.conflict(`Tag with slug "${body.slug}" already exists`);
    }

    // Insert new tag
    const id = uuid();
    await db.run(
      `INSERT INTO tags (id, name, slug, description, status, created_at, updated_at)
       VALUES (?, ?, ?, ?, ?, ?, ?)`,
      [
        id,
        body.name,
        body.slug,
        body.description ?? null,
        'active',
        now(),
        now(),
      ]
    );

    // Return the created tag
    const [tag] = await db.query<any>('SELECT * FROM tags WHERE id = ?', [id]);
    return c.json(tag, 201);
  } catch (error) {
    if (error instanceof ApiError) throw error;
    throw ApiError.internalServerError('Failed to create tag');
  }
});

export { publicApp as publicTags, adminApp as adminTags };
```

To register these routes, add to `apps/merchant/src/index.ts`:

```typescript
// Add these lines to the appropriate sections in index.ts
app.route('/v1/tags', publicTags);     // Public routes
app.route('/v1/tags', adminTags);      // Admin routes
```
