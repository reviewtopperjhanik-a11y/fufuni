## How to Add a New API Route

### File Structure

Create a new file at `apps/merchant/src/routes/my-feature.ts`. Follow the established pattern:

```typescript
// apps/merchant/src/routes/my-feature.ts
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, type HonoEnv } from '../types';
import { getDb } from '../db';
import { authMiddleware, adminOnly } from '../middleware/auth';
import { MyFeatureResponse, CreateFeatureBody } from '../schemas';
```

### Minimal Boilerplate

Every route file requires:

1. **Public router** (no auth):
```typescript
const publicApp = new OpenAPIHono<HonoEnv>();
```

2. **Admin router** (with auth):
```typescript
const adminApp = new OpenAPIHono<HonoEnv>();
adminApp.use('*', authMiddleware);
```

3. **Route definition** using `createRoute`:
```typescript
const exampleRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['MyFeature'],
  summary: 'Example endpoint',
  responses: {
    200: {
      content: { 'application/json': { schema: MyFeatureResponse } },
      description: 'Success',
    },
    500: {
      content: { 'application/json': { schema: ErrorResponse } },
      description: 'Server error',
    },
  },
});

publicApp.openapi(exampleRoute, async (c) => {
  return c.json({ message: 'Hello World' });
});
```

### Public vs Admin Endpoints

Split endpoints between two routers:

```typescript
// Public endpoint (no auth)
publicApp.openapi(getPublicRoute, async (c) => { /* ... */ });

// Admin endpoint (requires auth)
adminApp.openapi(
  createAdminRoute,
  adminOnly,
  async (c) => { /* ... */ }
);
```

Export both routers at the end:
```typescript
export { publicApp as publicMyFeature, adminApp as adminMyFeature };
```

### Database Access

Use `getDb()` to interact with SQLite:

```typescript
// Query data
const db = getDb(c.var.db);
const results = await db.query<{ id: string }>(
  'SELECT id FROM my_table WHERE status = ?',
  ['active']
);

// Run mutations
await db.run(
  'INSERT INTO my_table (id, name) VALUES (?, ?)',
  [uuid(), 'New Item']
);
```

### Error Handling

Use `ApiError` for consistent error responses:

```typescript
try {
  // Database operation
} catch (error) {
  if (error instanceof ApiError) throw error;
  throw ApiError.internalServerError('Operation failed');
}

// Common error types:
throw ApiError.notFound('Item not found');
throw ApiError.badRequest('Invalid input');
throw ApiError.conflict('Duplicate entry');
```

### Route Registration

In `apps/merchant/src/index.ts`:

1. Import your route:
```typescript
import { publicMyFeature, adminMyFeature } from './routes/my-feature';
```

2. Register public routes BEFORE auth middleware:
```typescript
app.route('/v1/my-feature', publicMyFeature);
```

3. Register admin routes AFTER auth middleware:
```typescript
app.route('/v1/my-feature', adminMyFeature);
```

### RBAC Guards

Available middleware from `../middleware/auth`:

| Guard | Usage |
|-------|-------|
| `adminOnly` | Admin dashboard access |
| `superAdminOnly` | Super admin privileges |
| `databaseAdminOnly` | Database management |
| `aiAccessOnly` | AI feature access |
| `mailAccessOnly` | Email system access |
| `validJwtAuthOnly` | Valid JWT required |

Example usage:
```typescript
adminApp.openapi(
  createRoute({ /* ... */ }),
  adminOnly,  // Apply guard
  async (c) => { /* ... */ }
);
```

### Complete Example: Tags Endpoint

Here's a full implementation of a tags management system:

```typescript
// apps/merchant/src/routes/tags.ts
import { OpenAPIHono, createRoute } from '@hono/zod-openapi';
import { z } from 'zod';
import { ApiError, uuid, type HonoEnv } from '../types';
import { getDb } from '../db';
import { authMiddleware, adminOnly } from '../middleware/auth';
import {
  TagResponse,
  TagListResponse,
  CreateTagBody,
  TagIdParam,
  ErrorResponse,
} from '../schemas';

// Public router
const publicApp = new OpenAPIHono<HonoEnv>();

// Public: List all tags
const listTagsRoute = createRoute({
  method: 'get',
  path: '/',
  tags: ['Tags'],
  summary: 'List all tags',
  responses: {
    200: {
      content: { 'application/json': { schema: TagListResponse } },
      description: 'List of tags',
    },
  },
});

publicApp.openapi(listTagsRoute, async (c) => {
  const db = getDb(c.var.db);
  try {
    const tags = await db.query<{ id: string; name: string }>(
      'SELECT id, name FROM tags ORDER BY name ASC'
    );
    return c.json({ items: tags });
  } catch (error) {
    throw ApiError.internalServerError('Failed to fetch tags');
  }
});

// Admin router
const adminApp = new OpenAPIHono<HonoEnv>();
adminApp.use('*', authMiddleware);

// Admin: Create tag
const createTagRoute = createRoute({
  method: 'post',
  path: '/',
  tags: ['Tags'],
  summary: 'Create a tag',
  security: [{ bearerAuth: ['admin:store'] }],
  middleware: [adminOnly] as const,
  request: {
    body: {
      content: { 'application/json': { schema: CreateTagBody } },
    },
  },
  responses: {
    201: {
      content: { 'application/json': { schema: TagResponse } },
      description: 'Tag created',
    },
    400: {
      content: { 'application/json': { schema: ErrorResponse } },
      description: 'Invalid input',
    },
  },
});

adminApp.openapi(createTagRoute, async (c) => {
  const body = c.req.valid('json');
  const db = getDb(c.var.db);

  try {
    const [existing] = await db.query<{ id: string }>(
      'SELECT id FROM tags WHERE name = ?',
      [body.name]
    );
    if (existing) {
      throw ApiError.conflict(`Tag "${body.name}" already exists`);
    }

    const id = uuid();
    await db.run(
      'INSERT INTO tags (id, name, description) VALUES (?, ?, ?)',
      [id, body.name, body.description || null]
    );

    const [tag] = await db.query<any>(
      'SELECT * FROM tags WHERE id = ?',
      [id]
    );
    return c.json(tag, 201);
  } catch (error) {
    if (error instanceof ApiError) throw error;
    throw ApiError.internalServerError('Failed to create tag');
  }
});

// Admin: Delete tag
const deleteTagRoute = createRoute({
  method: 'delete',
  path: '/:id',
  tags: ['Tags'],
  summary: 'Delete a tag',
  security: [{ bearerAuth: ['admin:store'] }],
  middleware: [adminOnly] as const,
  request: {
    params: TagIdParam,
  },
  responses: {
    200: {
      content: { 'application/json': { schema: z.object({ ok: z.boolean() }) } },
      description: 'Tag deleted',
    },
  },
});

adminApp.openapi(deleteTagRoute, async (c) => {
  const { id } = c.req.valid('param');
  const db = getDb(c.var.db);

  try {
    await db.run('DELETE FROM tags WHERE id = ?', [id]);
    return c.json({ ok: true });
  } catch (error) {
    throw ApiError.internalServerError('Failed to delete tag');
  }
});

export { publicApp as publicTags, adminApp as adminTags };
```

To register these routes in `index.ts`:

```typescript
import { publicTags, adminTags } from './routes/tags';

// Register before auth middleware
app.route('/v1/tags', publicTags);

// Register after auth middleware
app.route('/v1/tags', adminTags);
```
