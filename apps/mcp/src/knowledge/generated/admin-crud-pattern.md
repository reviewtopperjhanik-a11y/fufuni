## Admin CRUD Pattern Guide

### The Three-Component Pattern
Every admin CRUD page follows a consistent composition:

```tsx
// 1. useAdminCrud() — state + API logic
const { items, displayedItems, ... } = useAdminCrud<Entity>({ ... });

// 2. AdminCrudLayout — shell with header, filters, table
return (
  <AdminCrudLayout
    title="Products"
    addLabel="Add product"
    onAdd={openCreate}
    globalFilter={globalFilter}
    onGlobalFilterChange={setGlobalFilter}
    statusFilter={statusFilter}
    onStatusFilterChange={setStatusFilter}
  >
    {/* Table using displayedItems */}
  </AdminCrudLayout>
);

// 3. RowActions — per-row edit/delete buttons
const columns = [
  { accessorKey: "name", header: "Name" },
  {
    id: "actions",
    cell: ({ row }) => (
      <RowActions
        onEdit={() => openEdit(row.original)}
        onDelete={() => remove(row.original.id)}
      />
    ),
  },
];
```

### Delete Confirmation
The component embeds a HeroUI `Popover` for confirmation:
```tsx
<RowActions
  onEdit={() => openEdit(item)}
  onDelete={() => {
    if (confirm("Delete this item?")) {
      remove(item.id);
    }
  }}
/>
```

### Edit Flow
1. Clicking edit triggers `onEdit` → `openEdit(item)` from `useAdminCrud`.
2. The hook sets `isEditMode = true` and `editingItem = item`.
3. The modal opens with form pre-populated via `editingItem`.

## Modal Pattern

### Create vs Edit Distinction
```tsx
<Modal isOpen={isModalOpen} onClose={closeModal}>

  </Modal.Header>

          defaultValue={editingItem?.name || ""}
        />
      </TextField>
    </form>
  </Modal.Body>

```

### Form Reset Logic
```tsx
useEffect(() => {
  if (!isModalOpen) {
    reset(); // Clear form on close
  }
}, [isModalOpen]);

useEffect(() => {
  if (editingItem) {
    reset(editingItem); // Populate form on edit
  }
}, [editingItem]);
```

## Form Validation

### React-Hook-Form + Zod Integration
```tsx
import { zodResolver } from "@hookform/resolvers/zod";
import { z } from "zod";

const schema = z.object({
  name: z.string().min(1, "Required"),
  slug: z.string().min(1, "Required"),
});

type FormValues = z.infer<typeof schema>;

const { register, handleSubmit, reset, formState: { errors } } = useForm<FormValues>({
  resolver: zodResolver(schema),
});

const onSubmit = async (data: FormValues) => {
  if (isEditMode) {
    await update(editingItem!.id, data);
  } else {
    await create(data);
  }
  closeModal();
};
```

## Complete Example: ProductTags Admin Page

### Page Component
```tsx
// apps/client/src/pages/admin/product-tags.tsx
import { useAdminCrud } from "@/hooks/use-admin-crud";
import { AdminCrudLayout } from "@/shared/ui/admin/admin-crud-layout";
import { RowActions } from "@/shared/ui/admin/row-actions";
import { Table } from "@heroui/react";
import { useForm } from "react-hook-form";
import { z } from "zod";
import { zodResolver } from "@hookform/resolvers/zod";
import { api } from "@/lib/api";

const schema = z.object({
  name: z.string().min(1),
  slug: z.string().min(1),
});

type FormValues = z.infer<typeof schema>;

export function ProductTagsPage() {
  const {
    items,
    setItems,
    displayedItems,
    globalFilter,
    setGlobalFilter,
    statusFilter,
    setStatusFilter,
    isModalOpen,
    setIsModalOpen,
    isEditMode,
    editingItem,
    openCreate,
    openEdit,
    closeModal,
  } = useAdminCrud<ProductTag>({
    filterFn: (tag, term) =>
      tag.name.toLowerCase().includes(term) ||
      tag.slug.toLowerCase().includes(term),
  });

  const { register, handleSubmit, reset } = useForm<FormValues>({
    resolver: zodResolver(schema),
  });

  const fetchTags = async () => {
    const tags = await api.productTags.list();
    setItems(tags);
  };

  const createTag = async (data: FormValues) => {
    await api.productTags.create(data);
    await fetchTags();
  };

  const updateTag = async (id: string, data: FormValues) => {
    await api.productTags.update(id, data);
    await fetchTags();
  };

  const deleteTag = async (id: string) => {
    await api.productTags.delete(id);
    await fetchTags();
  };

  useEffect(() => { fetchTags(); }, []);

  const columns = [
    { accessorKey: "name", header: "Name" },
    { accessorKey: "slug", header: "Slug" },
    {
      id: "actions",
      cell: ({ row }) => (
        <RowActions
          onEdit={() => openEdit(row.original)}
          onDelete={() => deleteTag(row.original.id)}
        />
      ),
    },
  ];

  return (
    <>
      <AdminCrudLayout
        title="Product Tags"
        addLabel="Add tag"
        onAdd={openCreate}
        globalFilter={globalFilter}
        onGlobalFilterChange={setGlobalFilter}
        statusFilter={statusFilter}
        onStatusFilterChange={setStatusFilter}
      >
        <Table data={displayedItems} columns={columns} />
      </AdminCrudLayout>

      <Modal isOpen={isModalOpen} onClose={closeModal}>
        <Modal.Header>
          {isEditMode ? "Edit Tag" : "Create Tag"}
        </Modal.Header>
        <Modal.Body>
          <form onSubmit={handleSubmit((data) => {
            isEditMode
              ? updateTag(editingItem!.id, data)
              : createTag(data);
            closeModal();
          })}>
            <TextField>
              <Label>Name</Label>
              <Input
                {...register("name")}
                defaultValue={editingItem?.name || ""}
              />
            </TextField>
            <TextField>
              <Label>Slug</Label>
              <Input
                {...register("slug")}
                defaultValue={editingItem?.slug || ""}
              />
            </TextField>
            <Button type="submit" variant="primary">
              {isEditMode ? "Update" : "Create"}
            </Button>
          </form>
        </Modal.Body>
      </Modal>
    </>
  );
}
```

### API Client
```tsx
// apps/client/src/lib/api/product-tags.ts
export const productTags = {
  list: () => api.get<ProductTag[]>("/admin/product-tags"),
  create: (data: Omit<ProductTag, "id">) =>
    api.post("/admin/product-tags", data),
  update: (id: string, data: Partial<ProductTag>) =>
    api.patch(`/admin/product-tags/${id}`, data),
  delete: (id: string) => api.delete(`/admin/product-tags/${id}`),
};
```
